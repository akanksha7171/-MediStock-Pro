# MediStock Pro

Pharmacy stock, expiry and billing system. React (Vite + Tailwind) frontend, Express backend, MySQL database.

## Run on Windows (first time)
1. Install **Node.js LTS** (nodejs.org) and **MySQL** (or XAMPP). Remember your MySQL root password.
2. Open Command Prompt in the project folder and create the database:
   `mysql -u root -p < database\schema.sql`
3. Backend:
   `cd server` -> `copy .env.example .env` -> open `.env` in Notepad and fill in your MySQL password, a long random JWT_SECRET, ADMIN_EMAIL and ADMIN_PASSWORD (keep the password in double quotes) -> `npm install` -> `npm run seed` -> `npm start`
4. Frontend (second Command Prompt): `cd client` -> `npm install` -> `npm run dev`
5. Open http://localhost:5173 and sign in with the ADMIN_EMAIL / ADMIN_PASSWORD from your `.env`.

## What works (tested against a real database through the API)
- Login (JWT, bcrypt), owner/staff roles, staff accounts (create, disable), change password API
- Medicines: add, search (name/generic/barcode), archive. Edit exists in the API only.
- Batches with expiry status; stock adjustment API with reason and audit log
- Purchases (transactional, creates batches + stock movements; duplicate batch numbers rejected)
- Billing/POS: server re-checks expiry and stock, row locking, duplicate-submit protection, rollback on failure, tax calculated on the server
- Sales history with search/filters, full-sale return (stock restored, audit + movement records)
- Dashboard (live stats, 7-day chart), notifications (read/unread), analytics API, CSV export (sales, inventory, purchases; owner only)

## Not finished
- Pages not built yet: Analytics page, Settings page, Profile page (their APIs exist)
- No medicine edit form, no stock-adjustment form, no discount field in POS
- No PDF invoice (browser Print only), no partial returns, no demand forecast, no assistant, no barcode camera
- The UI was only compile-checked: it has NOT been viewed in a browser, so layout, mobile and visual polish are unverified
- shadcn/ui is not used (Tailwind + Framer Motion + Recharts + Lucide only)

## API (all except login/health need `Authorization: Bearer <token>`; (owner) = admin only)
auth/login, auth/change-password, me, users (owner), medicines, batches, stock/adjust (owner), suppliers, purchases (owner create), sales, sales/:id, sales/:id/return (owner), sales-search, dashboard, dashboard/trend, analytics, notifications, notifications/read-all, notifications/:id/read, settings (owner edit), export/:sales|inventory|purchases (owner)

CREATE DATABASE IF NOT EXISTS medistock_pro CHARACTER SET utf8mb4;
USE medistock_pro;
CREATE TABLE users(id INT AUTO_INCREMENT PRIMARY KEY,name VARCHAR(100) NOT NULL,email VARCHAR(150) NOT NULL UNIQUE,password_hash VARCHAR(255) NOT NULL,role ENUM('admin','staff') NOT NULL DEFAULT 'staff',is_active TINYINT(1) DEFAULT 1,created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP);
CREATE TABLE medicines(id INT AUTO_INCREMENT PRIMARY KEY,name VARCHAR(150) NOT NULL,generic_name VARCHAR(150),brand VARCHAR(100),category VARCHAR(80),dosage_form VARCHAR(50),strength VARCHAR(50),manufacturer VARCHAR(120),barcode VARCHAR(60) UNIQUE,description TEXT,requires_prescription TINYINT(1) DEFAULT 0,min_stock INT NOT NULL DEFAULT 10,tax_percent DECIMAL(5,2) NOT NULL DEFAULT 12.00,is_archived TINYINT(1) DEFAULT 0,created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,INDEX(name),INDEX(generic_name),INDEX(category));
CREATE TABLE suppliers(id INT AUTO_INCREMENT PRIMARY KEY,company_name VARCHAR(150) NOT NULL,contact_person VARCHAR(100),phone VARCHAR(20),email VARCHAR(150),address TEXT,gstin VARCHAR(20),notes TEXT,is_archived TINYINT(1) DEFAULT 0,created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP);
CREATE TABLE purchases(id INT AUTO_INCREMENT PRIMARY KEY,supplier_id INT NOT NULL,reference_no VARCHAR(50) NOT NULL UNIQUE,total_amount DECIMAL(12,2) NOT NULL,status ENUM('received','cancelled') DEFAULT 'received',created_by INT NOT NULL,created_at DATETIME DEFAULT CURRENT_TIMESTAMP,FOREIGN KEY(supplier_id) REFERENCES suppliers(id),FOREIGN KEY(created_by) REFERENCES users(id),INDEX(created_at));
CREATE TABLE batches(id INT AUTO_INCREMENT PRIMARY KEY,medicine_id INT NOT NULL,supplier_id INT,purchase_id INT,batch_no VARCHAR(50) NOT NULL,purchase_price DECIMAL(10,2) NOT NULL,selling_price DECIMAL(10,2) NOT NULL,received_qty INT NOT NULL,available_qty INT NOT NULL,mfg_date DATE,expiry_date DATE NOT NULL,created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,CONSTRAINT chk_qty CHECK(available_qty>=0 AND available_qty<=received_qty),UNIQUE(medicine_id,batch_no),FOREIGN KEY(medicine_id) REFERENCES medicines(id),FOREIGN KEY(supplier_id) REFERENCES suppliers(id),FOREIGN KEY(purchase_id) REFERENCES purchases(id),INDEX(expiry_date));
CREATE TABLE purchase_items(id INT AUTO_INCREMENT PRIMARY KEY,purchase_id INT NOT NULL,medicine_id INT NOT NULL,batch_id INT NOT NULL,quantity INT NOT NULL,purchase_price DECIMAL(10,2) NOT NULL,selling_price DECIMAL(10,2) NOT NULL,expiry_date DATE NOT NULL,FOREIGN KEY(purchase_id) REFERENCES purchases(id) ON DELETE CASCADE,FOREIGN KEY(medicine_id) REFERENCES medicines(id),FOREIGN KEY(batch_id) REFERENCES batches(id));
CREATE TABLE sales(id INT AUTO_INCREMENT PRIMARY KEY,invoice_no VARCHAR(30) NOT NULL UNIQUE,idempotency_key VARCHAR(64) UNIQUE,customer_name VARCHAR(100),subtotal DECIMAL(12,2) NOT NULL,discount DECIMAL(12,2) NOT NULL DEFAULT 0,tax DECIMAL(12,2) NOT NULL,total DECIMAL(12,2) NOT NULL,payment_method ENUM('cash','upi','card','other') NOT NULL,status ENUM('completed','returned','cancelled') DEFAULT 'completed',created_by INT NOT NULL,created_at DATETIME DEFAULT CURRENT_TIMESTAMP,FOREIGN KEY(created_by) REFERENCES users(id),INDEX(created_at));
CREATE TABLE sale_items(id INT AUTO_INCREMENT PRIMARY KEY,sale_id INT NOT NULL,medicine_id INT NOT NULL,batch_id INT NOT NULL,quantity INT NOT NULL,unit_price DECIMAL(10,2) NOT NULL,unit_cost DECIMAL(10,2) NOT NULL,discount DECIMAL(10,2) NOT NULL DEFAULT 0,tax_percent DECIMAL(5,2) NOT NULL,tax_amount DECIMAL(10,2) NOT NULL,line_total DECIMAL(12,2) NOT NULL,FOREIGN KEY(sale_id) REFERENCES sales(id),FOREIGN KEY(medicine_id) REFERENCES medicines(id),FOREIGN KEY(batch_id) REFERENCES batches(id));
CREATE TABLE stock_movements(id INT AUTO_INCREMENT PRIMARY KEY,batch_id INT NOT NULL,medicine_id INT NOT NULL,type ENUM('purchase','sale','return','adjustment') NOT NULL,quantity_change INT NOT NULL,reason VARCHAR(255),reference VARCHAR(50),user_id INT,created_at DATETIME DEFAULT CURRENT_TIMESTAMP,FOREIGN KEY(batch_id) REFERENCES batches(id),FOREIGN KEY(medicine_id) REFERENCES medicines(id),INDEX(created_at));
CREATE TABLE notifications(id INT AUTO_INCREMENT PRIMARY KEY,category ENUM('low_stock','expiring','expired','purchase','sale','inventory') NOT NULL,message VARCHAR(255) NOT NULL,dedupe_key VARCHAR(100) UNIQUE,is_read TINYINT(1) DEFAULT 0,created_at DATETIME DEFAULT CURRENT_TIMESTAMP);
CREATE TABLE returns(id INT AUTO_INCREMENT PRIMARY KEY,sale_id INT NOT NULL,reason VARCHAR(255),refund_amount DECIMAL(12,2) NOT NULL,created_by INT NOT NULL,created_at DATETIME DEFAULT CURRENT_TIMESTAMP,FOREIGN KEY(sale_id) REFERENCES sales(id),FOREIGN KEY(created_by) REFERENCES users(id));
CREATE TABLE return_items(id INT AUTO_INCREMENT PRIMARY KEY,return_id INT NOT NULL,sale_item_id INT NOT NULL,quantity INT NOT NULL,FOREIGN KEY(return_id) REFERENCES returns(id) ON DELETE CASCADE,FOREIGN KEY(sale_item_id) REFERENCES sale_items(id));
CREATE TABLE audit_logs(id INT AUTO_INCREMENT PRIMARY KEY,user_id INT,action VARCHAR(80) NOT NULL,entity VARCHAR(50),entity_id INT,details JSON,created_at DATETIME DEFAULT CURRENT_TIMESTAMP,INDEX(created_at));
CREATE TABLE pharmacy_settings(id INT PRIMARY KEY DEFAULT 1,name VARCHAR(150) DEFAULT 'MediStock Pharmacy',address TEXT,phone VARCHAR(20),gstin VARCHAR(20),expiry_critical_days INT DEFAULT 30,expiry_warning_days INT DEFAULT 90);
